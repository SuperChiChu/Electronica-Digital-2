#include <Bluepad32.h>

ControllerPtr joycon = nullptr;

HardwareSerial STM32Serial(2);

#define STM32_RX_PIN 16
#define STM32_TX_PIN 17

#define CTRL_LEFT   0x01
#define CTRL_RIGHT  0x02
#define CTRL_ACCEL  0x04
#define CTRL_BRAKE  0x08

#define STICK_DEADZONE 120

uint8_t lastControl = 0xFF;
uint32_t lastSend = 0;

void onConnectedController(ControllerPtr ctl)
{
    joycon = ctl;

    Serial.print("PLAYER 2 conectado: ");
    Serial.println(ctl->getModelName());
}

void onDisconnectedController(ControllerPtr ctl)
{
    if (joycon == ctl)
    {
        joycon = nullptr;
    }

    Serial.println("PLAYER 2 desconectado");
}

void setup()
{
    Serial.begin(115200);

    STM32Serial.begin(
        115200,
        SERIAL_8N1,
        STM32_RX_PIN,
        STM32_TX_PIN
    );

    BP32.setup(
        &onConnectedController,
        &onDisconnectedController
    );

    BP32.enableVirtualDevice(false);
}

void loop()
{
    BP32.update();

    if (millis() - lastSend < 20)
        return;

    lastSend = millis();

    uint8_t controls = 0;

    if (joycon != nullptr &&
        joycon->isConnected() &&
        joycon->isGamepad())
    {
        int32_t x = joycon->axisX();

        if (x < -STICK_DEADZONE)
            controls |= CTRL_LEFT;

        else if (x > STICK_DEADZONE)
            controls |= CTRL_RIGHT;

        if (joycon->r1())
            controls |= CTRL_ACCEL;

        if (joycon->l1())
            controls |= CTRL_BRAKE;

        if (controls != lastControl)
        {
            Serial.print("X = ");
            Serial.print(x);

            Serial.print(" | CTRL = 0x");

            if (controls < 0x10)
                Serial.print("0");

            Serial.println(controls, HEX);

            lastControl = controls;
        }
    }

    STM32Serial.write(controls);
}
